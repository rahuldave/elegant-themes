    library.py
    from dataclasses import dataclass

@dataclass(frozen=True)
class Book:
    title: str
    author: str
    year: int

def find_books(books: list[Book], query: str) -> list[Book]:
    """Find a title or author, ignoring case."""
    needle = query.casefold()
    return [book for book in books if needle in book.title.casefold()]

books = [Book("A Room with a View", "E. M. Forster", 1908)]
print(find_books(books, "room"))
